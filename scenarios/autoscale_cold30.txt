# Diurnal load with target-utilization autoscaling and a 30 s cold start: the treatment of demo 20.
# The controller reads the delayed views every 10 s and holds mean running/max_batch at 0.7, at most
# 32 replicas per decision; a turned-up replica serves nothing for 30 s. The finding is what that
# cold start costs in SLO during the day's rise, and what the drain costs on its way down.
name = autoscale_cold30
seed = 20260906
duration_s = 400
warmup_s = 40

# Fleet. A 70B-class model on 8xH100 per replica; step_base_ms is calibrated in
# bench/validate_epochs.py against a published batch-1 measurement.
replicas = 256
# The autoscaler's signal is running / max_batch per ready replica, so max_batch has to be the
# concurrency at which a replica is actually full. At this workload a replica's rated throughput,
# 7.3 rps, is about 30 sequences in flight by Little's law, so the count cap is set where the
# prefill budget binds anyway; the base fleet's 256 would make 0.7 of it a queue of thousands.
max_batch = 32
kv_capacity_tokens = 1370000
# Weight read for a 70B model on 8xH100 at 70% bandwidth utilization, 7.46 ms, plus 2.75 ms of fixed
# overhead. With the bandwidth term below this reproduces bench/validate_epochs.py to within 0.05 ms.
step_base_ms = 10.2
step_per_seq_ms = 0.0
step_per_kv_ktoken_ms = 0.0175
prefill_tokens_per_s = 28286
step_token_budget = 1024
max_queue = 400

# Workload. A mixture, not a single distribution: interactive chat plus long-document work in the
# same queue. The base fleet's 560 rps, which at a batch of 32 is 0.49 of the 256-replica fleet's
# rated 1152: inside the autoscaler's dead band, so the day's rise is what moves it first.
arrival_rps = 560
prompt_mean = 1200
prompt_cv = 1.2
output_mean = 300
output_cv = 1.5
long_probability = 0.08
long_prompt_mean = 24000
long_output_mean = 400

# The day: a 0.005 Hz sine is a 200 s day, and 0.6 of amplitude swings offered load from 224 to
# 896 rps, so at the 0.7 target the peak needs about 285 replicas and the trough about 70. The fixed
# fleet's rated 1152 covers the peak, so the control pays in idle replicas rather than in SLO.
perturbation = sine
perturb_amplitude = 0.6
perturb_frequency_hz = 0.005

routing = p2c
telemetry_interval_ms = 1000
telemetry_delay_ms = 200

client_timeout_s = 60
max_attempts = 1
ttft_slo_ms = 2000
itl_slo_ms = 80
e2e_slo_s = 60
sample_interval_ms = 250

autoscaling = target_utilization
autoscale_target = 0.7
autoscale_interval_s = 10
# 32 per decision rather than the default 8, so the step is not what bounds the rise and the cold
# start is what the comparison shows.
autoscale_step = 32
autoscale_cooldown_s = 30
min_replicas = 64
max_replicas = 384
warmup_delay_s = 30
drain_timeout_s = 60
