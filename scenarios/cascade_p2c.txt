# Dynamic 10 of docs/scope-today.md: affinity hotspot failover cascade (needs dynamics 9 and 4).
# Affinity concentrates a root's traffic on the replicas holding its prefix. When a holder dies, its
# roots' traffic lands on replicas with no cached prefix, every request pays the full prefill, and the
# inheriting replicas' step time and queues spike. When the holder returns at t=90 s it starts cold
# (crash() clears the prefix cache, U27a), so it is a hotspot without a cache until it re-earns hits.
#
# The two cascade_* scenarios are affinity_sticky.txt / affinity_off.txt plus one failure line, same
# seed, so a difference in outcome is a difference in routing. Replica 0 is the crashed one because it
# is the measured hottest holder in demo 14's affinity_sticky export: highest mean queued depth over
# t in [30, 60] s (0.13 against a fleet mean of 0.02) and highest running count (29.8 against 10.5);
# replicas 0-5 form the holder band. In p2c the same replica is an ordinary member of the fleet.
#
# The prefix topology is a [GUESS] per docs/calibration.md section 9.1: 100 roots of 800 tokens, Zipf
# s = 1.0 popularity, 10 % of requests forking an existing session, a prefix cache of about 30 % of
# each replica's KV. Nothing here is measured; sweep it before believing a number from it.
# cascade_p2c: the control. p2c holds no prefixes on purpose, so losing replica 0 loses 1/256 of the
# fleet and nothing else.
name = cascade_p2c
seed = 20260906
duration_s = 120
warmup_s = 15

# Fleet. A 70B-class model on 8xH100 per replica; step_base_ms is calibrated in
# bench/validate_epochs.py against a published batch-1 measurement.
replicas = 256
# Sequence cap and token budget both present, because which one binds is the point. At a ~3.2k mean
# context the token budget allows about 420 sequences, so the count cap binds for ordinary traffic
# and the token budget binds once long prompts arrive.
max_batch = 256
kv_capacity_tokens = 1370000
# Weight read for a 70B model on 8xH100 at 70% bandwidth utilization, 7.46 ms, plus 2.75 ms of fixed
# overhead. With the bandwidth term below this reproduces bench/validate_epochs.py to within 0.05 ms.
step_base_ms = 10.2
step_per_seq_ms = 0.0
step_per_kv_ktoken_ms = 0.0175
prefill_tokens_per_s = 28286
step_token_budget = 1024
max_queue = 400

# Workload. A mixture, not a single distribution: interactive chat plus long-document work in the
# same queue. The mixture is what creates head-of-line blocking, so a unimodal load would hide the
# phenomenon being studied.
arrival_rps = 560
prompt_mean = 1200
prompt_cv = 1.2
output_mean = 300
output_cv = 1.5
long_probability = 0.08
long_prompt_mean = 24000
long_output_mean = 400

# Prefix topology (identical to the three affinity_* scenarios).
prefix_roots = 100
prefix_root_tokens = 800
prefix_zipf_s = 1.0
session_fork_rate = 0.1
prefix_cache_tokens = 400000

# One holder crashes at t=60 s and returns empty at t=90 s, in crash.txt's syntax.
failures = t=60,replica=0,kind=crash,until=90

routing = p2c
# Fresh telemetry, unlike route_p2c.txt's 1000 / 200 ms, and shared by all three. Measured at the
# baseline's staleness: every ratio from 0.5 to 2.2 herds to a load CV of about 1.2 and loses to
# p2c on every metric, because a holder receives a second of a root's traffic before the view
# that would refuse it arrives. The ceiling can only act on a view fresh enough to see the load it
# is capping, so the staleness interaction is a finding of its own (dynamic 2), not of this knob.
telemetry_interval_ms = 100
telemetry_delay_ms = 10

client_timeout_s = 60
max_attempts = 1
ttft_slo_ms = 2000
itl_slo_ms = 80
e2e_slo_s = 60
sample_interval_ms = 250
